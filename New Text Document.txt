    <p>Find a location or enter coordinates to generate an image with embedded GPS EXIF data.</p>

            <label for="lonInput">Longitude (Decimal):</label>
            <input type="text" id="lonInput" placeholder="e.g., 2.2945">
        </div>
        <button onclick="updateFromCoords()">Update Map</button>

        <button id="generateBtn" onclick="generateImage()" disabled>Generate Image with EXIF</button>

            <button id="downloadBtn">Download Image</button>
        </a>

    <script>
        let currentLat = null;
        let currentLon = null;

        // Initialize OpenLayers Map
        const map = new ol.Map({
            target: 'map',
            layers: [
                new ol.layer.Tile({
                    source: new ol.source.OSM()
                })
            ],
            view: new ol.View({
                center: ol.proj.fromLonLat([0, 0]),
                zoom: 2
            })
        });

        // Add a vector layer for the marker
        const markerSource = new ol.source.Vector();
        const markerLayer = new ol.layer.Vector({
            source: markerSource,
            style: new ol.style.Style({
                image: new ol.style.Circle({
                    radius: 7,
                    fill: new ol.style.Fill({color: 'red'}),
                    stroke: new ol.style.Stroke({color: 'white', width: 2})
                })
            })
        });
        map.addLayer(markerLayer);

        // Allow clicking on map to set coordinates
        map.on('click', function(evt) {
            const coords = ol.proj.toLonLat(evt.coordinate);
            setCoordinates(coords[1], coords[0]);
        });


        // Tab logic
        function openTab(evt, tabName) {
            let i, tabcontent, tablinks;
            tabcontent = document.getElementsByClassName("tabcontent");
            for (i = 0; i < tabcontent.length; i++) {
                tabcontent[i].style.display = "none";
            }
            tablinks = document.getElementsByClassName("tablinks");
            for (i = 0; i < tablinks.length; i++) {
                tablinks[i].className = tablinks[i].className.replace(" active", "");
            }
            document.getElementById(tabName).style.display = "block";
            evt.currentTarget.className += " active";
        }

        // --- Location Fetching ---

        async function searchOSM() {
            const query = document.getElementById('searchQuery').value;
            const errorDiv = document.getElementById('searchError');
            errorDiv.innerText = '';

            if (!query) return;

            try {
                // Use Nominatim API for geocoding
                const response = await fetch(`https://nominatim.openstreetmap.org/search?q=${encodeURIComponent(query)}&format=json&limit=1`);
                const data = await response.json();

                if (data && data.length > 0) {
                    const lat = parseFloat(data[0].lat);
                    const lon = parseFloat(data[0].lon);
                    setCoordinates(lat, lon);
                } else {
                    errorDiv.innerText = 'Location not found.';
                }
            } catch (err) {
                errorDiv.innerText = 'Error fetching data from OSM.';
                console.error(err);
            }
        }

        function updateFromCoords() {
            const latStr = document.getElementById('latInput').value;
            const lonStr = document.getElementById('lonInput').value;
            const errorDiv = document.getElementById('coordError');
            errorDiv.innerText = '';

            const lat = parseFloat(latStr);
            const lon = parseFloat(lonStr);

            if (isNaN(lat) || isNaN(lon) || lat < -90 || lat > 90 || lon < -180 || lon > 180) {
                errorDiv.innerText = 'Invalid coordinates. Lat: -90 to 90, Lon: -180 to 180.';
                return;
            }
            setCoordinates(lat, lon);
        }

        function setCoordinates(lat, lon) {
            currentLat = lat;
            currentLon = lon;
            
            // Update UI
            document.getElementById('currentSelection').innerText = `${lat.toFixed(6)}, ${lon.toFixed(6)}`;
            document.getElementById('latInput').value = lat.toFixed(6);
            document.getElementById('lonInput').value = lon.toFixed(6);
            document.getElementById('generateBtn').disabled = false;

            // Update Map
            const projectedCoords = ol.proj.fromLonLat([lon, lat]);
            map.getView().animate({ center: projectedCoords, zoom: 14 });
            
            markerSource.clear();
            markerSource.addFeature(new ol.Feature(new ol.geom.Point(projectedCoords)));
        }

        // --- Image Generation & EXIF ---

        // Helper to convert decimal degrees to rational format needed by EXIF
        function degToDmsRational(deg) {
            const d = Math.abs(deg);
            const degrees = Math.floor(d);
            const minutes = Math.floor((d - degrees) * 60);
            const seconds = Math.round((d - degrees - minutes / 60) * 3600 * 100) / 100;
            return [[degrees, 1], [minutes, 1], [Math.round(seconds * 100), 100]];
        }

        function generateImage() {
            if (currentLat === null || currentLon === null) return;

            // 1. Create a dummy image using Canvas
            const canvas = document.createElement('canvas');
            canvas.width = 400;
            canvas.height = 300;
            const ctx = canvas.getContext('2d');
            
            // Draw something simple
            ctx.fillStyle = '#eee';
            ctx.fillRect(0, 0, canvas.width, canvas.height);
            ctx.fillStyle = '#333';
            ctx.font = '20px sans-serif';
            ctx.fillText('Geotagged Dummy Image', 20, 50);
            ctx.font = '14px sans-serif';
            ctx.fillText(`Lat: ${currentLat.toFixed(6)}`, 20, 100);
            ctx.fillText(`Lon: ${currentLon.toFixed(6)}`, 20, 125);

            // Get base64 JPEG
            const jpegData = canvas.toDataURL('image/jpeg', 0.9);

            // 2. Build the EXIF data structure
            const zeroth = {};
            const exif = {};
            const gps = {};

            // Required basic tags
            zeroth[piexif.ImageIFD.Make] = "WebApp Generator";
            
            // Format GPS Data
            const latRef = currentLat < 0 ? 'S' : 'N';
            const lonRef = currentLon < 0 ? 'W' : 'E';

            gps[piexif.GPSIFD.GPSLatitudeRef] = latRef;
            gps[piexif.GPSIFD.GPSLatitude] = degToDmsRational(currentLat);
            gps[piexif.GPSIFD.GPSLongitudeRef] = lonRef;
            gps[piexif.GPSIFD.GPSLongitude] = degToDmsRational(currentLon);
            gps[piexif.GPSIFD.GPSVersionID] = [2, 2, 0, 0];

            const exifObj = {"0th": zeroth, "Exif": exif, "GPS": gps};

            // 3. Insert EXIF into the JPEG
            const exifBytes = piexif.dump(exifObj);
            const newDataUrl = piexif.insert(exifBytes, jpegData);

            // 4. Update UI to show and download the result
            const imgElement = document.getElementById('outputImage');
            imgElement.src = newDataUrl;
            
            const downloadLink = document.getElementById('downloadLink');
            downloadLink.href = newDataUrl;

            document.getElementById('result-container').style.display = 'block';
        }
    </script>
